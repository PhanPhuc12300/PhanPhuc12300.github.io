from datetime import date from js import document today = date.today() target = date(2025, 9, 20) if today >= target: result = "Hôm nay là ngày 20/9/2025 hoặc sau ngày đó." else: result = "Hôm nay vẫn trước ngày 20/9/2025." # Xuất kết quả ra div#output document.getElementById("output").innerText = result
